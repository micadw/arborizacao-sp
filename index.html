Observatório da Arborização Urbana de São Paulo

Árvores, cobertura vegetal, orçamento e metas da cidade — com dados abertos

Projeto pessoal e acadêmico, construído só com dados abertos; não representa posição oficial da Prefeitura de São Paulo. Extração de 2026-11-20. Veja Metas, Orçamento e as fontes.

Código 
orc2026 = new Map(orc.filter(d => d.ano === 2026).map(d => [d.chave_sub, d]))
camada = {
  const fc = structuredClone(nivel === "distrito" ? distritos : subs);
  for (const f of fc.features) {
    const p = f.properties;
    p.nome = p.nm_distrito_municipal ?? p.nm_subprefeitura;
    if (nivel === "distrito") p.alvo_meta85 = +(p.pct_cobertura_vegetal + 1).toFixed(1);
    else {
      const o = orc2026.get(p.chave_sub);
      p.liquidado_mi = o ? +(o.Vl_Liquidado / 1e6).toFixed(1) : null;
      p.atualizado_mi = o ? +(o.Vl_Orcado_Atualizado / 1e6).toFixed(1) : null;
      p.pct_liquidado = o ? o.pct_liquidado : null;
    }
  }
  return fc;
}
rotulo = [...indicadores].find(([, v]) => v === indicador)[0]

ficha = {
  const f = sel;
  if (!f) return html`<div class="ficha"><em>Passe o mouse sobre o mapa.</em></div>`;
  const p = f.properties;
  const linhas = nivel === "distrito"
    ? [["Subprefeitura (código)", p.cd_identificador_subprefeitura], ["Cobertura vegetal", fmt(p.pct_cobertura_vegetal, 1) + "%"],
       ["Alvo Meta 85 (2028)", fmt(p.alvo_meta85, 1) + "%"], ["Cobertura arbórea", fmt(p.pct_cobertura_arborea, 1) + "%"],
       ["Árvores viárias", fmt(p.arvores_viarias)], ["Árvores por km²", fmt(p.arvores_por_km2)]]
    : [["Árvores viárias", fmt(p.arvores_viarias)], ["Quedas 2026", fmt(p.quedas_registradas)],
       ["Quedas / 10 mil árvores", fmt(p.quedas_por_10mil_arvores, 1)], ["Orçamento 2026 atualizado", "R$ " + fmt(p.atualizado_mi, 1) + " mi"],
       ["Liquidado", "R$ " + fmt(p.liquidado_mi, 1) + " mi (" + fmt(p.pct_liquidado, 1) + "%)"]];

}
